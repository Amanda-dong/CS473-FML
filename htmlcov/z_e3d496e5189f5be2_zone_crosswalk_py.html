79def _load_all_nta_codes() -> list[str]: 

80 """Load all NYC 2020 NTA codes from local GeoJSON when available.""" 

81 for candidate in ("data/raw/nta.geojson", "data/raw/nta2020_nyc.geojson"): 

82 geojson_path = Path(candidate) 

83 if not geojson_path.exists(): 

84 continue 

85 try: 

86 payload = json.loads(geojson_path.read_text(encoding="utf-8")) 

87 except Exception: 

88 continue 

89 features = payload.get("features", []) 

90 codes: set[str] = set() 

91 for feature in features: 

92 props = feature.get("properties", {}) if isinstance(feature, dict) else {} 

93 nta = str(props.get("nta2020", "")).strip() 

94 if nta: 

95 codes.add(nta) 

96 return sorted(codes) 

97 return []

145def _build_zone_to_nta() -> dict[str, list[str]]: 

146 """Return crosswalk that preserves base zones and fills all remaining NTAs.""" 

147 mapping = {zone_id: list(ntas) for zone_id, ntas in _BASE_ZONE_TO_NTA.items()} 

148 covered = {nta for ntas in mapping.values() for nta in ntas} 

149 for nta in _load_all_nta_codes(): 

150 if nta in covered: 

151 continue 

152 mapping[_generic_zone_id(nta)] = [nta] 

153 return mapping

158# Reverse lookup: NTA code -> list of zone IDs 

159NTA_TO_ZONES: dict[str, list[str]] = {} 

160for _zone, _ntas in ZONE_TO_NTA.items():

173def resolve_nta_to_zone_id(nta: str | None) -> str | None: 

174 """Map an ACS NTA code (e.g. ``MN0202``) to one micro-zone ``zone_id``. 

175 

176 Returns ``None`` if the NTA is not part of :data:`ZONE_TO_NTA` (e.g. a 

177 NYC block outside the modeled micro-zone list). 

178 """ 

179 if nta is None or (isinstance(nta, float) and pd.isna(nta)): 

180 return None 

181 code = str(nta).strip().upper() 

182 if not code: 

183 return None 

184 zones = NTA_TO_ZONES.get(code) 

185 if not zones: 

186 return None 

187 if len(zones) == 1: 

188 return zones[0] 

189 return NTA_PRIMARY_ZONE.get(code, sorted(zones)[0])

192def aggregate_nta_to_zone( 

193 nta_df: pd.DataFrame, 

194 zone_col: str = "nta_id", 

195 agg_rules: dict[str, str] | None = None, 

196 weights_col: str | None = None, 

197) -> pd.DataFrame: 

198 """Aggregate NTA-level data to zone-level using the crosswalk. 

199 

200 Parameters 

201 ---------- 

202 nta_df: 

203 DataFrame with an NTA identifier column and optionally a year/time column. 

204 zone_col: 

205 Name of the column containing NTA codes. 

206 agg_rules: 

207 Mapping of column name -> aggregation function (e.g. {"population": "sum", 

208 "median_income": "mean"}). Numeric columns not listed default to "mean". 

209 weights_col: 

210 Optional column for population/sample-weighted aggregation. When 

211 provided, numeric columns use weighted mean instead of simple mean 

212 (unless overridden by ``agg_rules``). 

213 

214 Returns 

215 ------- 

216 DataFrame with ``zone_id`` replacing the NTA column, aggregated per zone (and 

217 per ``year``/``time_key`` if present). 

218 """ 

219 if nta_df.empty or zone_col not in nta_df.columns: 

220 return pd.DataFrame() 

221 

222 # Build exploded mapping frame 

223 rows = [] 

224 for zone_id, nta_list in ZONE_TO_NTA.items(): 

225 for nta in nta_list: 

226 rows.append({"zone_id": zone_id, zone_col: nta}) 

227 mapping = pd.DataFrame(rows) 

228 

229 # Backward-compat: also map 4-char prefixes of 6-char codes so ETL data 

230 # using 2010 NTA codes (BK02) can join against 2020 codes (BK0202 -> prefix BK02). 

231 # We deduplicate so one 4-char prefix maps to the first zone found. 

232 seen_4char: set[str] = set() 

233 extra_rows = [] 

234 for zone_id, nta_list in ZONE_TO_NTA.items(): 

235 for nta in nta_list: 

236 if len(nta) == 6: 

237 prefix = nta[:4] 

238 if prefix not in seen_4char: 

239 seen_4char.add(prefix) 

240 extra_rows.append({"zone_id": zone_id, zone_col: prefix}) 

241 if extra_rows: 

242 mapping = pd.concat([mapping, pd.DataFrame(extra_rows)], ignore_index=True) 

243 

244 extra_2010 = [] 

245 for nta_2010, nta_2020 in _NTA_2010_TO_2020.items(): 

246 for zone_id, nta_list in ZONE_TO_NTA.items(): 

247 if nta_2020 in nta_list: 

248 extra_2010.append({"zone_id": zone_id, zone_col: nta_2010}) 

249 break 

250 if extra_2010: 

251 mapping = pd.concat([mapping, pd.DataFrame(extra_2010)], ignore_index=True) 

252 mapping = mapping.drop_duplicates(subset=[zone_col, "zone_id"]) 

253 

254 merged = nta_df.merge(mapping, on=zone_col, how="inner") 

255 if merged.empty: 

256 return pd.DataFrame() 

257 

258 # Determine groupby keys 

259 time_col = ( 

260 "year" 

261 if "year" in merged.columns 

262 else ("time_key" if "time_key" in merged.columns else None) 

263 ) 

264 group_keys = ["zone_id"] + ([time_col] if time_col else []) 

265 

266 # Build aggregation dict 

267 numeric_cols = merged.select_dtypes(include=["number"]).columns.tolist() 

268 skip_cols = set(group_keys) | ({weights_col} if weights_col else set()) 

269 default_agg = {c: "mean" for c in numeric_cols if c not in skip_cols} 

270 if agg_rules: 

271 default_agg.update(agg_rules) 

272 

273 if not default_agg: 

274 return merged[group_keys].drop_duplicates().reset_index(drop=True) 

275 

276 # Weighted aggregation when weights_col is available 

277 if weights_col and weights_col in merged.columns: 

278 import numpy as _np 

279 

280 def _weighted_agg(grp: pd.DataFrame) -> pd.Series: 

281 w = grp[weights_col].values.astype(float) 

282 w_sum = w.sum() 

283 out = {} 

284 for col, func in default_agg.items(): 

285 if col not in grp.columns: 

286 continue 

287 if func == "mean" and w_sum > 0: 

288 out[col] = float(_np.dot(w, grp[col].values.astype(float)) / w_sum) 

289 elif func == "sum": 

290 out[col] = float(grp[col].sum()) 

291 else: 

292 out[col] = float(grp[col].agg(func)) 

293 return pd.Series(out) 

294 

295 result = ( 

296 merged.groupby(group_keys) 

297 .apply(_weighted_agg, include_groups=False) 

298 .reset_index() 

299 ) 

300 else: 

301 result = merged.groupby(group_keys, as_index=False).agg(default_agg) 

302 if time_col and time_col != "time_key": 

303 result = result.rename(columns={time_col: "time_key"}) 

304 return result
